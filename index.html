# agent_alpha.py # Execution guard + agent loop. Defaults to paper trading. import logging from dataclasses import dataclass, field from datetime import datetime, timezone from enum import Enum logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s") log = logging.getLogger("agent-alpha") class Decision(str, Enum): BUY = "BUY"; SELL = "SELL"; HOLD = "HOLD" class Volatility(str, Enum): LOW = "Low"; MODERATE = "Moderate"; EXTREME = "Extreme" @dataclass class Signal: origin: str # e.g. "Momentum-EMA" symbol: str decision: Decision entry: float stop_loss: float take_profit: float reasoning: str @dataclass class RiskConfig: max_risk_per_trade: float = 0.02 # 2% of capital min_reward_risk: float = 2.0 # matches the 1:2 rule max_leverage: float = 5.0 paper_trading: bool = True @dataclass class Account: equity: float free_margin: float open_positions: list = field(default_factory=list) def classify_volatility(atr_pct: float) -> Volatility: # atr_pct = ATR / price * 100 (computed upstream) if atr_pct < 1.0: return Volatility.LOW if atr_pct < 3.0: return Volatility.MODERATE return Volatility.EXTREME class PreTradeGuard: """Hard checks enforced in code. The LLM cannot override these.""" def __init__(self, cfg: RiskConfig): self.cfg = cfg def check(self, sig: Signal, acct: Account, qty: float, leverage: float, vol: Volatility) -> tuple[bool, str, float]: if sig.decision == Decision.HOLD: return False, "HOLD: no order", 0.0 if sig.stop_loss <= 0: return False, "REJECT: stop-loss not set", 0.0 risk_per_unit = abs(sig.entry - sig.stop_loss) if risk_per_unit == 0: return False, "REJECT: invalid stop distance", 0.0 reward_per_unit = abs(sig.take_profit - sig.entry) if reward_per_unit / risk_per_unit < self.cfg.min_reward_risk: return False, "REJECT: reward/risk below 1:2", 0.0 if leverage > self.cfg.max_leverage: return False, f"REJECT: leverage {leverage}x exceeds {self.cfg.max_leverage}x", 0.0 # Extreme volatility: halve size automatically if vol == Volatility.EXTREME: qty *= 0.5 # Cap risk at max_risk_per_trade of equity max_loss = acct.equity * self.cfg.max_risk_per_trade max_qty = max_loss / risk_per_unit qty = min(qty, max_qty) notional = qty * sig.entry if notional / leverage > acct.free_margin: return False, "REJECT: insufficient free margin", 0.0 return True, f"OK (vol={vol.value}, qty={qty:.6f})", qty class PaperBroker: """Replace with real broker API calls (e.g. ccxt) when going live.""" def place_order(self, sig: Signal, qty: float) -> dict: return {"id": f"paper-{datetime.now(timezone.utc).timestamp():.0f}", "symbol": sig.symbol, "side": sig.decision.value, "qty": qty, "entry": sig.entry, "sl": sig.stop_loss, "tp": sig.take_profit} class AgentAlpha: def __init__(self, cfg: RiskConfig, broker, account: Account): self.cfg, self.broker, self.account = cfg, broker, account self.guard = PreTradeGuard(cfg) def log_decision(self, sig: Signal, status: str, risk: str): ts = datetime.now(timezone.utc).isoformat() log.info("ts=%s origin=%s decision=%s reason=%s | status=%s | risk=%s", ts, sig.origin, sig.decision.value, sig.reasoning, status, risk) def handle_signal(self, sig: Signal, atr_pct: float, qty: float, leverage: float): vol = classify_volatility(atr_pct) ok, msg, final_qty = self.guard.check(sig, self.account, qty, leverage, vol) self.log_decision(sig, "EXECUTED" if ok else "BLOCKED", msg) if not ok: return None if self.cfg.paper_trading: log.info("PAPER mode: simulated order") order = self.broker.place_order(sig, final_qty) self.account.open_positions.append(order) return order def on_candle_close(self, positions_prices: dict): """Call once per closed candle: move SL to breakeven after TP1, enforce invalidation.""" for pos in self.account.open_positions: price = positions_prices.get(pos["symbol"]) if price is None: continue # Breakeven logic would go here once TP1 is hit (track TP1 separately) log.info("monitor %s price=%s sl=%s tp=%s", pos["symbol"], price, pos["sl"], pos["tp"]) if __name__ == "__main__": acct = Account(equity=10_000, free_margin=8_000) agent = AgentAlpha(RiskConfig(), PaperBroker(), acct) sig = Signal("Momentum-EMA", "BTC/USDT", Decision.BUY, entry=64250, stop_loss=63600, take_profit=65540, reasoning="EMA 9 crossed above EMA 21 on 1h, volume confirmed") agent.handle_signal(sig, atr_pct=1.4, qty=0.2, leverage=2)
